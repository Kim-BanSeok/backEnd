async function worker(hold = false) {

  const ready = await new Promise((resolve, reject) => {
    const timer = setTimeout(() => reject(new Error('Worker startup timeout')), 15000);
    child.once('error', reject); child.once('exit', code => reject(new Error(`Worker exited ${code}`)));
    child.on('message', message => {
      if (message.event === 'ready') { clearTimeout(timer); resolve(message); }
      if (message.event === 'started') starts++;
      if (message.id) { pending.get(message.id)?.(message); pending.delete(message.id); }
    });
  });
  const instance = { ...ready, child, started: () => starts, command: action => new Promise(resolve => { const id = randomUUID(); pending.set(id, resolve); child.send({ id, action }); }) };
  workers.push(instance); return instance;
}
