async function request(method, route, { body, token, form } = {}) {
  const response = await fetch(origin + '/api/v1' + route, { method,
    headers: { ...(body !== undefined && { 'Content-Type': 'application/json' }), ...(token && { Authorization: `Bearer ${token}` }) },
    body: form ?? (body !== undefined ? JSON.stringify(body) : undefined) });
  const json = await response.json();
  traces.push({ method, route, authenticated: Boolean(token), request: redact(body), response: { status: response.status, body: redact(json) } });
  return { status: response.status, json };
}
